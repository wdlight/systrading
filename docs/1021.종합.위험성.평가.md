### 7.1 백엔드 수정 (CRITICAL)

**파일 1**: `backend/app/services/chart_cache_service.py`

```python
import fcntl
from contextlib import contextmanager

@contextmanager
def file_lock(file_path: Path, mode: str = 'r'):
    """파일 잠금 컨텍스트 매니저"""
    with open(file_path, mode, encoding="utf-8") as f:
        try:
            fcntl.flock(f.fileno(), fcntl.LOCK_EX)  # 배타적 잠금
            yield f
        finally:
            fcntl.flock(f.fileno(), fcntl.LOCK_UN)  # 잠금 해제

def _save_to_cache(self, stock_code: str, date: datetime, candles: List[ChartCandle]):
    """캐시 저장 (파일 잠금 추가)"""
    cache_file = self._get_cache_file_path(stock_code, date)
    data = [candle.model_dump() for candle in candles]

    # ✅ 파일 잠금으로 동시 접근 방지
    with file_lock(cache_file, 'w') as f:
        json.dump(data, f, ensure_ascii=False, indent=2)

def _load_from_cache(self, stock_code: str, date: datetime):
    """캐시 읽기 (파일 잠금 추가)"""
    cache_file = self._get_cache_file_path(stock_code, date)

    # ✅ 읽기도 잠금 필요 (쓰기 중 읽기 방지)
    with file_lock(cache_file, 'r') as f:
        data = json.load(f)
        return [ChartCandle(**item) for item in data]
```

```python
        try:
            target_date = datetime.fromisoformat(candle.timestamp)
            success = await trading_service.update_minute_candle(
                stock_code=stock_code,
                target_date=target_date,
                candle_data=candle
            )

            if success:
                logger.info(f"✅ 분봉 저장 성공 (시도 {attempt + 1})")
                return  # 성공 시 즉시 종료
            else:
                logger.warning(f"⚠️ 저장 실패 (시도 {attempt + 1})")

        except Exception as e:
            logger.error(f"❌ 저장 오류 (시도 {attempt + 1}): {e}")

        # 재시도 대기 (exponential backoff)
        if attempt < max_retries - 1:
            await asyncio.sleep(retry_delay * (attempt + 1))

    # 모든 재시도 실패
    logger.critical(f"🚨 [CRITICAL] 모든 재시도 실패: {stock_code} {candle.timestamp}")
    # TODO: Dead-letter queue 또는 알림 시스템 연동
```
