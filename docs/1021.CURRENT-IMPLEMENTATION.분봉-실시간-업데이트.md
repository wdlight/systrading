```python
async def _handle_tick_data(self, message: Dict[str, Any]):
    """체결 데이터 처리 및 분봉 집계"""

    # 1. 데이터 추출
    data = message.get("data", {})
    stock_code = message.get("stock_code")
    price = float(data.get("price", 0))
    executed_time = data.get("executed_time")  # "140523"
    trade_volume = int(data.get("trade_volume", 0))
    acc_volume = int(data.get("acc_volume", 0))

    # 2. 타임스탬프 파싱
    executed_ts = parse_kis_time(executed_time)
    minute_key = executed_ts.strftime("%Y-%m-%dT%H:%M:00")
    # 예: "2025-10-21T14:05:00"

    # 3. 장 시간 체크
    if not TradingHoursManager.is_trading_hours(executed_ts_naive):
        return  # 장 외 데이터 무시

    # 4. 분봉 상태 갱신
    async with self.minute_state_lock:
        state = self.minute_candle_state.get(stock_code)

        # 새로운 분봉인가?
        if state and state.minute_key != minute_key:
            await self._flush_closed_candle(stock_code, state)  # 이전 분봉 완료
            state = None

        # 분봉 상태 생성/업데이트
        if not state:
            state = MinuteCandleState(
                minute_key=minute_key,
                open=price,      # 첫 체결가
                high=price,
                low=price,
                close=price,
                volume=trade_volume,
                start_ts=executed_ts,
                last_tick_ts=executed_ts,
            )
            self.minute_candle_state[stock_code] = state
        else:
            # 기존 분봉에 체결 추가
            state.apply_tick(price, trade_volume, executed_ts)
            # → high/low 갱신, close 갱신, volume 누적

    # 5. 실시간 브로드캐스트
    await self._broadcast_minute_snapshot(stock_code, state)
```

```python
async def _flush_closed_candle(self, stock_code: str, state: MinuteCandleState):
    """완료된 분봉을 캐시/DB에 저장"""

    # 1. 최종 캔들 생성
    final_candle = ChartCandle(
        timestamp=state.minute_key,
        open=state.open,
        high=state.high,
        low=state.low,
        close=state.close,
        volume=state.volume,
        is_final=True,  # ← 완료된 분봉!
    )

    # 2. 최종 브로드캐스트
    await self.connection_manager.broadcast({
        "type": "minute_candle_finalized",  # ← 완료 이벤트
        "stock_code": stock_code,
        "candle": final_candle.dict(),
    })

    # 3. 영구 저장 (캐시 파일 업데이트)
    if self._minute_persist_handler:
        await self._minute_persist_handler(stock_code, final_candle)

    # 4. 로그
    logger.info(
        f"📊 [MINUTE-FINALIZED] {stock_code} {state.minute_key} | "
        f"O={state.open} H={state.high} L={state.low} C={state.close} V={state.volume}"
    )
```

## 📋 데이터 구조 정리

### MinuteCandleState (backend)

**파일**: `backend/app/models/realtime_minute.py`

```python
class MinuteCandleState:
    minute_key: str         # "2025-10-21T14:05:00"
    open: float             # 시가
    high: float             # 고가
    low: float              # 저가
    close: float            # 종가 (현재가)
    volume: int             # 거래량
    start_ts: datetime      # 첫 체결 시각
    last_tick_ts: datetime  # 마지막 체결 시각

    def apply_tick(self, price: float, volume: int, ts: datetime):
        """체결 데이터 적용"""
        self.high = max(self.high, price)
        self.low = min(self.low, price)
        self.close = price
        self.volume += volume
        self.last_tick_ts = ts
```

## ⚠️ 현재 구현의 한계점

### 1. H0STCNI0는 "내 주문 체결"만 받음

**문제**:
- H0STCNI0 = 내가 낸 주문의 체결 통보
- 시장 전체의 체결 데이터가 아님!

**영향**:
- 내가 주문을 안 내면 데이터가 안 옴
- 따라서 분봉 집계가 안 됨

**해결책**:
- ❌ **현재**: H0STCNI0 (내 주문만)
- ✅ **필요**: H0STAS0 or 다른 실시간 시세 TR_ID

---

### 2. 장 시간에만 작동

**파일**: `realtime_service.py:460-462`

```python
if not TradingHoursManager.is_trading_hours(executed_ts_naive, include_extended=True):
    logger.debug(f"장 시간 외 체결 데이터 무시")
    return
```

**영향**:
- 09:00-15:30 외에는 분봉 생성 안 됨
- 장 전후 시간외 거래 데이터는 `include_extended=True`로 포함 가능
