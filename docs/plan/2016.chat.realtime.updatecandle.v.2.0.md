🧪 5. FastAPI 실전 코드
📁 ohlc_store.py
from collections import defaultdict
from datetime import datetime

ohlc_store = defaultdict(lambda: {
    "open": None, "high": 0, "low": float("inf"), "close": None, "volume": 0
})

def update_ohlc(price, volume):
    now = datetime.now().strftime('%Y-%m-%d %H:%M')
    ohlc = ohlc_store[now]
    if ohlc["open"] is None:
        ohlc["open"] = price
    ohlc["high"] = max(ohlc["high"], price)
    ohlc["low"] = min(ohlc["low"], price)
    ohlc["close"] = price
    ohlc["volume"] += volume

📁 main.py
from fastapi import FastAPI
import requests, asyncio, websockets, json
from ohlc_store import ohlc_store, update_ohlc
from datetime import datetime

app = FastAPI()

ACCESS_TOKEN = "<ACCESS_TOKEN>"
APP_KEY = "<APP_KEY>"
APP_SECRET = "<APP_SECRET>"

@app.get("/ohlc/init")
def get_init_ohlc():
    url = "https://openapi.koreainvestment.com:9443/uapi/domestic-stock/v1/quotations/inquire-time-itemchartprice"
    headers = {
        "authorization": f"Bearer {ACCESS_TOKEN}",
        "appkey": APP_KEY,
        "appsecret": APP_SECRET,
        "tr_id": "FHKST03010200",
        "custtype": "P"
    }
    params = {
        "fid_cond_mrkt_div_code": "J",
        "fid_input_iscd": "005930",
        "fid_input_hour_1": "090000",
        "fid_input_hour_2": "153000",
        "fid_period_div_code": "1",
        "fid_org_adj_prc": "0"
    }
    res = requests.get(url, headers=headers, params=params).json()
    bars = []
    for item in res["output2"]:
        time = datetime.strptime(f"{item['stck_bsop_date']} {item['stck_cntg_hour']}", "%Y%m%d %H%M%S")
        bars.append({
            "time": int(time.timestamp()),
            "open": int(item["stck_oprc"]),
            "high": int(item["stck_hgpr"]),
            "low": int(item["stck_lwpr"]),
            "close": int(item["stck_prpr"]),
            "volume": int(item["acml_vol"])
        })
    return bars

@app.get("/ohlc")
def get_ohlc():
    bars = []
    for t, o in ohlc_store.items():
        dt = datetime.strptime(t, "%Y-%m-%d %H:%M")
        bars.append({
            "time": int(dt.timestamp()),
            "open": o["open"],
            "high": o["high"],
            "low": o["low"],
            "close": o["close"],
            "volume": o["volume"]
        })
    return bars

async def websocket_worker():
    uri = "wss://openapi.koreainvestment.com:9443/websocket"
    headers = {
        "authorization": f"Bearer {ACCESS_TOKEN}",
        "appkey": APP_KEY,
        "appsecret": APP_SECRET,
    }
    async with websockets.connect(uri, extra_headers=headers) as ws:
        await ws.send(json.dumps({
            "header": headers,
            "body": {
                "tr_id": "H0TCNT0",
                "tr_key": "005930"
            }
        }))
        while True:
            msg = await ws.recv()
            fields = msg.split("^")
            price = int(fields[2])
            volume = 1
            update_ohlc(price, volume)

@app.on_event("startup")
async def startup_event():
    asyncio.create_task(websocket_worker())

import { widget } from '@tradingview/charting_library'
import { useEffect, useRef } from 'react'

export default function Chart() {
  const ref = useRef(null)

  useEffect(() => {
    const tv = new widget({
      symbol: '005930.KR',
      interval: '1',
      container: ref.current,
      library_path: '/static/charting_library/',
      datafeed: {
        onReady: (cb) => cb({ supported_resolutions: ['1', '5', '30'] }),
        getBars: async (_, __, ___, ____, callback) => {
          const res = await fetch("http://localhost:8000/ohlc/init")
          const data = await res.json()
          callback(data, { noData: false })
        },
        subscribeBars: (_, __, onRealtime) => {
          const interval = setInterval(async () => {
            const res = await fetch("http://localhost:8000/ohlc")
            const data = await res.json()
            const latest = data[data.length - 1]
            onRealtime(latest)
          }, 3000)
          return () => clearInterval(interval)
        },
        unsubscribeBars: () => {}
      }
    })

    return () => tv.remove()
  }, [])

  return <div ref={ref} style={{ height: '500px' }} />
}

📌 7. 주의 사항 요약

단일 종목에 대해 분봉 → 실시간 → 갱신까지 연결 구성 가능
