# yfinance 기반 지수 서비스 대체 계획

## 📋 개요
한투 API 의존성을 줄이기 위해 yfinance 라이브러리를 사용한 지수 서비스 대체 계획입니다.  
기존 endpoint와 동일한 파라미터를 유지하되, URL만 `yf-index`로 변경하여 간단히 교체할 수 있도록 설계합니다.

## 🎯 목표
- **한투 API 트래픽 감소**: 지수 조회 시 한투 API 호출 제한 회피
- **동일한 인터페이스**: 기존 클라이언트 코드 수정 없이 교체 가능
- **안정적인 서비스**: yfinance의 무료 서비스로 지속적인 데이터 제공
- **쉬운 구현**: 초보 개발자도 따라할 수 있는 단계별 가이드

```python
# backend/app/services/yfinance_service/yf_index_service.py
import yfinance as yf
from typing import Dict, Any, Optional
from datetime import datetime
import asyncio
from loguru import logger

class YFinanceIndexService:
    """yfinance 기반 지수 데이터 서비스"""
    
    def __init__(self):
        self.cache = {}
        self.cache_duration = 30  # 30초 캐시
        
    async def get_market_indices(self) -> Dict[str, Dict[str, Any]]:
        """주요 지수 데이터 조회"""
        try:
            # 지수 매핑 테이블
            index_mapping = {
                "kospi": "^KS11",      # KOSPI Composite Index
                "kosdaq": "^KQ11",     # Kosdaq Composite Index  
                "nasdaq": "^IXIC",     # NASDAQ Composite
                "dow": "^DJI",         # Dow Jones Industrial Average
                "sp500": "^GSPC",      # S&P 500
                "usd_krw": "USDKRW=X"  # USD/KRW 환율
            }
            
            indices = {}
            
            # 각 지수별로 데이터 조회
            for index_name, ticker in index_mapping.items():
                try:
                    data = await self._fetch_index_data(ticker)
                    if data:
                        indices[index_name] = self._format_index_data(index_name, ticker, data)
                except Exception as e:
                    logger.warning(f"{index_name} 데이터 조회 실패: {e}")
                    indices[index_name] = self._get_default_data(index_name, ticker)
            
            return indices
            
        except Exception as e:
            logger.error(f"지수 데이터 조회 실패: {e}")
            return self._get_all_default_data()
    
    async def _fetch_index_data(self, ticker: str) -> Optional[Dict]:
        """개별 지수 데이터 조회"""
        try:
            # 비동기로 yfinance 호출
            loop = asyncio.get_event_loop()
            ticker_obj = yf.Ticker(ticker)
            info = await loop.run_in_executor(None, ticker_obj.info)
            
            return info
        except Exception as e:
            logger.error(f"{ticker} 데이터 조회 실패: {e}")
            return None
    
    def _format_index_data(self, index_name: str, ticker: str, data: Dict) -> Dict[str, Any]:
        """yfinance 데이터를 기존 형식으로 변환"""
        current = data.get("regularMarketPrice", 0.0)
        previous_close = data.get("previousClose", current)
        change = current - previous_close if previous_close else 0.0
        change_rate = (change / previous_close * 100) if previous_close else 0.0
        
        # 기존 형식에 맞춰 변환
        return {
            "code": self._get_index_code(index_name),
            "market": self._get_market_code(index_name),
            "current": round(current, 2),
            "change": round(change, 2),
            "change_rate": round(change_rate, 2)
        }
    
    def _get_index_code(self, index_name: str) -> str:
        """지수 코드 매핑"""
        code_mapping = {
            "kospi": "0001",
            "kosdaq": "1001", 
            "nasdaq": "NDX",
            "dow": "DJI",
            "sp500": "US500",
            "usd_krw": "USDKRW"
        }
        return code_mapping.get(index_name, "UNKNOWN")
    
    def _get_market_code(self, index_name: str) -> str:
        """시장 코드 매핑"""
        market_mapping = {
            "kospi": "U",
            "kosdaq": "K",
            "nasdaq": "N", 
            "dow": "N",
            "sp500": "N",
            "usd_krw": "X"
        }
        return market_mapping.get(index_name, "U")
    
    def _get_default_data(self, index_name: str, ticker: str) -> Dict[str, Any]:
        """기본값 데이터 반환"""
        return {
            "code": self._get_index_code(index_name),
            "market": self._get_market_code(index_name),
            "current": 0.0,
            "change": 0.0,
            "change_rate": 0.0
        }
    
    def _get_all_default_data(self) -> Dict[str, Dict[str, Any]]:
        """모든 지수의 기본값 데이터"""
        indices = ["kospi", "kosdaq", "nasdaq", "dow", "sp500", "usd_krw"]
        return {name: self._get_default_data(name, "") for name in indices}
```

```python
# backend/app/api/yf_index.py
from fastapi import APIRouter, Depends, HTTPException
from typing import Dict, Any
from loguru import logger

from app.services.yfinance_service.yf_index_service import YFinanceIndexService
from app.models.schemas import MarketOverview

router = APIRouter(prefix="/yf-index", tags=["YFinance Index"])

# 전역 서비스 인스턴스
_yf_service = None

def get_yf_index_service() -> YFinanceIndexService:
    """YFinance 지수 서비스 인스턴스 반환"""
    global _yf_service
    if _yf_service is None:
        _yf_service = YFinanceIndexService()
    return _yf_service

@router.get("/overview", response_model=MarketOverview, summary="yfinance 시장 현황 조회")
async def get_yf_market_overview(
    yf_service: YFinanceIndexService = Depends(get_yf_index_service)
) -> Dict[str, Any]:
    """
    yfinance를 사용하여 KOSPI, KOSDAQ, NASDAQ, DOW, S&P500, USD/KRW 지수 정보를 조회합니다.
    기존 /api/stocks/overview와 동일한 형식으로 응답합니다.
    """
    try:
        logger.info("yfinance 기반 시장 현황 조회 시작")
        
        # 지수 데이터 조회
        indices = await yf_service.get_market_indices()
        
        # 기존 형식에 맞춰 응답 구성
        response = {
            "market_status": "open",
            "kospi": indices.get("kospi", {}),
            "kosdaq": indices.get("kosdaq", {}),
            "nasdaq": indices.get("nasdaq", {}),
            "sp500": indices.get("sp500", {}),
            "dow": indices.get("dow", {}),  # DOW 추가
            "usd_krw": indices.get("usd_krw", {}),
            "top_gainers": [],  # 빈 배열 (향후 확장 가능)
            "top_losers": []    # 빈 배열 (향후 확장 가능)
        }
        
        logger.info("yfinance 기반 시장 현황 조회 완료")
        return response
        
    except Exception as e:
        logger.error(f"yfinance 시장 현황 조회 실패: {e}")
        raise HTTPException(status_code=500, detail=f"시장 현황 조회 중 오류가 발생했습니다: {str(e)}")

@router.get("/health", summary="yfinance 서비스 상태 확인")
async def health_check():
    """yfinance 서비스 상태 확인"""
    try:
        yf_service = get_yf_index_service()
        # 간단한 테스트 조회
        test_data = await yf_service._fetch_index_data("^KS11")
        if test_data:
            return {"status": "healthy", "message": "yfinance 서비스 정상 동작"}
        else:
            return {"status": "warning", "message": "yfinance 데이터 조회 실패"}
    except Exception as e:
        return {"status": "error", "message": f"yfinance 서비스 오류: {str(e)}"}
```

```python
async def _fetch_index_data_with_retry(self, ticker: str, max_retries: int = 3):
    for attempt in range(max_retries):
        try:
            return await self._fetch_index_data(ticker)
        except Exception as e:
            if attempt == max_retries - 1:
                logger.error(f"{ticker} 최대 재시도 횟수 초과: {e}")

            await asyncio.sleep(2 ** attempt)  # 지수 백오프
```

```python
# 문제: API 응답이 느림
# 해결: 캐싱 및 비동기 처리 최적화
class YFinanceIndexService:
    def __init__(self):
        self.cache = {}
        self.cache_duration = 60  # 캐시 시간 증가
    
    async def get_market_indices_cached(self):
        cache_key = "market_indices"
        now = time.time()
        
        # 캐시 확인
        if cache_key in self.cache:
            cached_time, data = self.cache[cache_key]
            if now - cached_time < self.cache_duration:
                return data
        
        # 새 데이터 조회
        data = await self.get_market_indices()
        self.cache[cache_key] = (now, data)
        return data
```

```python
# 문제: yfinance 데이터 형식 변경
# 해결: 유연한 데이터 파싱
def _format_index_data_safe(self, index_name: str, ticker: str, data: Dict) -> Dict[str, Any]:
    """안전한 데이터 형식 변환"""
    try:
        # 다양한 필드명 시도
        current = (data.get("regularMarketPrice") or 
                  data.get("currentPrice") or 
                  data.get("price") or 0.0)
        
        previous_close = (data.get("previousClose") or 
                         data.get("previousPrice") or 
                         current)
        
        change = current - previous_close if previous_close else 0.0
        change_rate = (change / previous_close * 100) if previous_close else 0.0
        
        return {
            "code": self._get_index_code(index_name),
            "market": self._get_market_code(index_name),
            "current": round(float(current), 2),
            "change": round(float(change), 2),
            "change_rate": round(float(change_rate), 2)
        }
    except Exception as e:
        logger.error(f"데이터 형식 변환 실패: {e}")
        return self._get_default_data(index_name, ticker)
```

```typescript
import { useState, useEffect, useCallback } from 'react';
import { MarketOverview, RegionalMarketData, MarketRegion } from '@/types/market';
import { fetchMarketOverview, fetchRegionalMarketData } from '@/lib/api';

export const useMarketData = () => {
  const [marketData, setMarketData] = useState<MarketOverview | null>(null);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState<string | null>(null);

  const loadMarketData = useCallback(async () => {
    try {
      setLoading(true);
      setError(null);
      const data = await fetchMarketOverview();
      setMarketData(data);
    } catch (err) {
      setError(err instanceof Error ? err.message : 'Failed to load market data');
    } finally {
      setLoading(false);
    }
  }, []);

  useEffect(() => {
    loadMarketData();
    
    // 60초마다 데이터 새로고침
    const interval = setInterval(loadMarketData, 60000);
    return () => clearInterval(interval);
  }, [loadMarketData]);

  return {
    marketData,
    loading,
    error,
    refetch: loadMarketData
  };
};
```

```typescript
export const useMarketData = () => {
  // 캐시 확인 로직 추가
```

#### 2. 에러 바운더리
```typescript
// src/components/ErrorBoundary.tsx
import React from 'react';

interface ErrorBoundaryState {
  hasError: boolean;
  error?: Error;
}

export class ErrorBoundary extends React.Component<
  React.PropsWithChildren<{}>,
  ErrorBoundaryState
> {
  constructor(props: React.PropsWithChildren<{}>) {
    super(props);
    this.state = { hasError: false };
  }

  static getDerivedStateFromError(error: Error): ErrorBoundaryState {
    return { hasError: true, error };
  }

  componentDidCatch(error: Error, errorInfo: React.ErrorInfo) {
    console.error('Market data error:', error, errorInfo);
  }

  render() {
    if (this.state.hasError) {
      return (
        <div className="error-boundary">
          <h2>Something went wrong with market data</h2>
          <button onClick={() => this.setState({ hasError: false })}>
            Try again
          </button>
        </div>
      );
    }

    return this.props.children;
  }
}
```
