#### 2. RealtimeDataService에 분봉 집계만 추가
```python
# realtime_service.py에 추가
class RealtimeDataService:
    def __init__(self, ...):
        # 기존 코드...
        self.minute_candle_cache: Dict[str, Dict[str, Any]] = {}  # 간단한 캐시
    
    async def _handle_tick_data(self, message: Dict[str, Any]):
        # 기존 tick_update 로직 유지...
        
        # 분봉 집계 추가 (최소한)
        data = message.get('data', {})
        stock_code = data.get('종목코드')
        price = float(data.get('현재가', 0))
        volume = int(data.get('체결거래량', 0))
        
        now = datetime.now()
        minute_key = now.strftime('%Y-%m-%dT%H:%M:00')
        
        if stock_code not in self.minute_candle_cache:
            self.minute_candle_cache[stock_code] = {}
            
        if minute_key not in self.minute_candle_cache[stock_code]:
            self.minute_candle_cache[stock_code][minute_key] = {
                'open': price, 'high': price, 'low': price, 'close': price, 'volume': 0
            }
        
        candle = self.minute_candle_cache[stock_code][minute_key]
        candle['high'] = max(candle['high'], price)
        candle['low'] = min(candle['low'], price) 
        candle['close'] = price
        candle['volume'] += volume
        
        # WebSocket 브로드캐스트
        await self.connection_manager.broadcast({
            "type": "minute_candle_update",
            "stock_code": stock_code,
            "data": {
                "timestamp": minute_key,
                **candle
            }
        })
```

### 프론트엔드 변경사항:

#### 1. 새로운 훅 생성 (단일 책임)
```typescript
// hooks/useRealtimeMinuteCandles.ts
export function useRealtimeMinuteCandles(stockCode: string, enabled: boolean = true) {
  const [currentCandle, setCurrentCandle] = useState<ChartCandle | null>(null);
  
  useEffect(() => {
    if (!enabled) return;
    
    const unsubscribe = subscribeToMessage('minute_candle_update', (message) => {
      if (message.stock_code === stockCode) {
        setCurrentCandle(message.data);
      }
    });
    
    return unsubscribe;
  }, [stockCode, enabled]);
  
  return currentCandle;
}
```

#### 2. 기존 차트 컴포넌트에서 활용
```typescript
// 차트 컴포넌트에서
const realtimeCandle = useRealtimeMinuteCandles(stockCode, timeframe === 'minute');
const historicalData = useHistoricalChartData({ stockCode });

// 마지막 캔들 업데이트 로직
const chartData = useMemo(() => {
  if (!realtimeCandle || historicalData.length === 0) return historicalData;
  
  const lastCandle = historicalData[historicalData.length - 1];
  const lastTime = lastCandle.timestamp.substring(0, 16);
  const realtimeTime = realtimeCandle.timestamp.substring(0, 16);
  
  if (lastTime === realtimeTime) {
    // 같은 분봉: 마지막 캔들 교체
    return [...historicalData.slice(0, -1), realtimeCandle];
  }
  
  return historicalData; // 다른 분봉이면 기존 폴링 로직 활용
}, [historicalData, realtimeCandle]);
```
